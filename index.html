    <!-- 📥 上傳區塊 -->

    <div class="card">

        <h3>📥 上傳新書籍 (EPUB)</h3>

        <h3>🔧 系統管理員進階設定</h3>

<script>

    // 網頁開啟時即刻檢查使用者登入狀態

    document.addEventListener("DOMContentLoaded", checkUserStatus);


    function checkUserStatus() {

        fetch('/user/status')

            .then(res => res.json())

            .then(data => {

                const authArea = document.getElementById('authArea');

                const adminPanel = document.getElementById('adminPanel');

                

                if (data.logged_in) {

                    authArea.innerHTML = `

                        <span>歡迎，<b>${data.username}</b> ${data.is_admin ? '<span class="tag tag-admin">管理員</span>' : ''}</span>

                        <button class="btn-danger" onclick="handleLogout()" style="margin-left:10px;">登出</button>

                    `;

                    // 如果是管理員，把後台設定面板秀出來

                    if (data.is_admin) adminPanel.style.display = 'block';

                } else {

                    adminPanel.style.display = 'none';

                    authArea.innerHTML = `

                        <input type="text" id="username" placeholder="帳號" style="width:100px;">

                        <input type="password" id="password" placeholder="密碼" style="width:100px;">

                        <button class="btn-primary" onclick="handleLogin()">登入</button>

                        <button class="btn-success" onclick="handleRegister()">註冊</button>

                    `;

                }

                // 每次狀態更新，刷新書籍清單

                loadBookList();

            });

    }


    // 🔐 會員系統操作邏輯

    function handleLogin() {

        const u = document.getElementById('username').value;

        const p = document.getElementById('password').value;

        fetch('/login', {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ username: u, password: p })

        }).then(res => {

            if (res.ok) { checkUserStatus(); } 

            else { alert("登入失敗，請檢查帳號密碼！"); }

        });

    }


    function handleRegister() {

        const u = document.getElementById('username').value;

        const p = document.getElementById('password').value;

        fetch('/register', {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ username: u, password: p })

        }).then(res => res.json()).then(data => {

            if (data.success) { alert("註冊成功！現在可以直接點擊登入。"); } 

            else { alert(data.error); }

        });

    }


    function handleLogout() {

        fetch('/logout', { method: 'POST' }).then(() => checkUserStatus());

    }


    // 📥 上傳檔案邏輯

    function handleUpload() {

        const fileInput = document.getElementById('epubFile');

        const statusDiv = document.getElementById('uploadStatus');

        if (!fileInput.files[0]) { alert("請先選擇一個 EPUB 檔案！"); return; }


        const formData = new FormData();

        formData.append('file', fileInput.files[0]);


        statusDiv.style.color = "orange";

        statusDiv.innerText = "正在上傳並解析書籍（這可能需要幾秒鐘）...";


        fetch('/upload', { method: 'POST', body: formData })

            .then(res => res.json())

            .then(data => {

                if (data.error) {

                    statusDiv.style.color = "var(--danger)";

                    statusDiv.innerText = `失敗: ${data.error}`;

                } else {

                    statusDiv.style.color = "var(--success)";

                    statusDiv.innerText = data.message;

                    loadBookList(); // 重新整理書櫃

                }

            }).catch(() => {

                statusDiv.style.color = "var(--danger)";

                statusDiv.innerText = "連線失敗，請檢查網路或伺服器紀錄。";

            });

    }


    // 📖 渲染書籍清單與下載功能

    function loadBookList() {

        const listDiv = document.getElementById('bookList');

        fetch('/books')

            .then(res => res.json())

            .then(books => {

                if (books.length === 0) {

                    listDiv.innerHTML = '<p style="color: #999;">書櫃空空如也，快去上傳一本吧！</p>';

                    return;

                }

                

                listDiv.innerHTML = books.map(book => `

                    <div class="book-item">

                        <div>

                            <strong style="font-size: 1.1em; color: var(--dark);">${book.title}</strong>

                            <span style="color: #777; margin-left: 10px;">分類櫃：${book.series_name}</span>

                            ${book.is_temporary ? '<span class="tag tag-temp">⏳ 24h 臨時</span>' : ''}

                            <br>

                            <small style="color: #999;">上傳者: ${book.uploader || '匿名訪客'}</small>

                        </div>

                        <div>

                            <!-- ⚡ 重點：直接呼叫後端 TXT 一鍵轉換導出接口 -->

                            <button class="btn-success" onclick="window.location.href='/books/${book.id}/download/txt'">

                                📥 下載 TXT 檔

                            </button>

                        </div>

                    </div>

                `).join('');

            });

    }


    // 👑 管理員功能：建立系列標籤

    function createNewSeries() {

        const nameInput = document.getElementById('newSeriesName');

        const statusDiv = document.getElementById('seriesStatus');

        const name = nameInput.value.trim();

        if(!name) return;


        fetch('/series', {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ name: name })

        }).then(res => res.json()).then(data => {

            if(data.error) {

                alert(data.error);

            } else {

                statusDiv.innerText = `成功建立分類標籤：「${data.name}」！`;

                nameInput.value = "";

                setTimeout(() => statusDiv.innerText = "", 3000);

            }

        });

    }
