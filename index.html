<script>

    // 網頁載入完成後立即檢查登入狀態

    document.addEventListener("DOMContentLoaded", checkUserStatus);


    function checkUserStatus() {

        fetch('/user/status')

            .then(res => res.json())

            .then(data => {

                const authArea = document.getElementById('authArea');

                const adminPanel = document.getElementById('adminPanel');

                

                if (data.logged_in) {

                    authArea.innerHTML = `

                        <span>歡迎，<b>${data.username}</b> ${data.is_admin ? '<span class="tag tag-admin">管理員</span>' : ''}</span>

                        <button class="btn-danger" onclick="handleLogout()" style="margin-left:10px;">登出</button>

                    `;

                    // 如果是管理員，開啟後台面板

                    if (data.is_admin) {

                        adminPanel.style.display = 'block';

                    } else {

                        adminPanel.style.display = 'none';

                    }

                } else {

                    adminPanel.style.display = 'none';

                    authArea.innerHTML = `

                        <input type="text" id="username" placeholder="帳號">

                        <input type="password" id="password" placeholder="密碼">

                        <button class="btn-primary" onclick="handleLogin()">登入</button>

                        <button class="btn-success" onclick="handleRegister()">註冊</button>

                    `;

                }

                // 狀態更新後，同步載入書籍清單

                loadBookList();

            }).catch(err => {

                console.error("無法連接到後端伺服器", err);

            });

    }


    // 🔐 會員系統機制

    function handleLogin() {

        const u = document.getElementById('username').value.trim();

        const p = document.getElementById('password').value.trim();

        if (!u || !p) { alert("請輸入帳號與密碼！"); return; }


        fetch('/login', {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ username: u, password: p })

        }).then(res => {

            if (res.ok) { 

                checkUserStatus(); 

            } else { 

                alert("登入失敗，請檢查您的帳號密碼是否正確！"); 

            }

        });

    }


    function handleRegister() {

        const u = document.getElementById('username').value.trim();

        const p = document.getElementById('password').value.trim();

        if (!u || !p) { alert("請輸入想要註冊的帳號與密碼！"); return; }


        fetch('/register', {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ username: u, password: p })

        }).then(res => res.json()).then(data => {

            if (data.success) { 

                alert("註冊成功！現在可以直接點擊「登入」按鈕。"); 

            } else { 

                alert(data.error || "註冊失敗，該帳號可能已被使用。"); 

            }

        });

    }


    function handleLogout() {

        fetch('/logout', { method: 'POST' }).then(() => checkUserStatus());

    }


    // 📥 上傳檔案機制

    function handleUpload() {

        const fileInput = document.getElementById('epubFile');

        const statusDiv = document.getElementById('uploadStatus');

        if (!fileInput.files[0]) { alert("請先選擇一個本地的 EPUB 檔案！"); return; }


        const formData = new FormData();

        formData.append('file', fileInput.files[0]);


        statusDiv.style.color = "orange";

        statusDiv.innerText = "⚡ 正在上傳並解析書籍（大型書籍需要較多時間，請勿重新整理網頁）...";


        fetch('/upload', { method: 'POST', body: formData })

            .then(res => res.json())

            .then(data => {

                if (data.error) {

                    statusDiv.style.color = "var(--danger)";

                    statusDiv.innerText = `❌ 失敗原因: ${data.error}`;

                } else {

                    statusDiv.style.color = "var(--success)";

                    statusDiv.innerText = `🎉 ${data.message}`;

                    fileInput.value = ""; // 清空上傳欄位

                    loadBookList(); // 重新整理書櫃

                }

            }).catch(() => {

                statusDiv.style.color = "var(--danger)";

                statusDiv.innerText = "❌ 連線失敗，請檢查後端日誌。";

            });

    }


    // 📖 渲染書籍清單與下載功能

    function loadBookList() {

        const listDiv = document.getElementById('bookList');

        fetch('/books')

            .then(res => res.json())

            .then(books => {

                if (!books || books.length === 0) {

                    listDiv.innerHTML = '<p style="color: #999; text-align: center; padding: 20px 0;">書櫃空空如也，快去上傳一本吧！</p>';

                    return;

                }

                

                listDiv.innerHTML = books.map(book => `

                    <div class="book-item">

                        <div>

                            <strong style="font-size: 1.1em; color: var(--dark);">${book.title}</strong>

                            <span style="color: #666; margin-left: 10px; font-size: 0.9em;">分類：${book.series_name}</span>

                            ${book.is_temporary ? '<span class="tag tag-temp">⏳ 24h 臨時</span>' : ''}

                            <br>

                            <small style="color: #999; font-size: 0.8em;">上傳者: ${book.uploader || '匿名訪客'}</small>

                        </div>

                        <div>

                            <!-- 📥 直接調用後端精準格式轉換導出接口 -->

                            <button class="btn-success" onclick="window.location.href='/books/${book.id}/download/txt'">

                                📥 一鍵導出 TXT

                            </button>

                        </div>

                    </div>

                `).join('');

            });

    }


    // 👑 管理員功能：建立系列分類櫃

    function createNewSeries() {

        const nameInput = document.getElementById('newSeriesName');

        const statusDiv = document.getElementById('seriesStatus');

        const name = nameInput.value.trim();

        if (!name) { alert("請輸入系列分類名稱！"); return; }


        fetch('/series', {

            method: 'POST',

            headers: { 'Content-Type': 'application/json' },

            body: JSON.stringify({ name: name })

        }).then(res => res.json()).then(data => {

            if (data.error) {

                alert(data.error);

            } else {

                statusDiv.innerText = `✅ 成功建立全局分類櫃：「${data.name}」！`;

                nameInput.value = "";

                setTimeout(() => { statusDiv.innerText = ""; }, 3000);

                loadBookList(); // 重新整理清單

            }

        });

    }
