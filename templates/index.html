    <script>

        let currentBooks = [];

        let userInfo = { logged_in: false, username: '' };

        let isRegisterMode = false;

        let activeCategory = '全部書籍';

        

        // 閱讀器變數

        let ePubBook = null;

        let rendition = null;


        window.addEventListener('DOMContentLoaded', () => {

            checkUserStatus();

            loadBooks();

            setupDragAndDrop();

        });


        async function checkUserStatus() {

            try {

                const res = await fetch('/user/status');

                if (res.ok) userInfo = await res.json();

            } catch (e) { console.warn(e); }

            finally { renderUserPanel(); }

        }


        function renderUserPanel() {

            const panel = document.getElementById('userPanel');

            if (userInfo.logged_in) {

                panel.innerHTML = `<span style="margin-right: 15px;">${userInfo.username}</span><button class="btn btn-outline" onclick="submitLogout()">登出</button>`;

            } else {

                panel.innerHTML = `<button class="btn btn-primary" onclick="showModal('authModal')">登入/註冊</button>`;

            }

        }


        // --- 📖 閱讀器邏輯 ---

        function openReader(fileUrl, title) {

            if(!fileUrl || fileUrl === "null") {

                alert("此書籍沒有實體檔案，請重新上傳！");

                return;

            }

            document.getElementById('readerTitle').innerText = title.replace('.epub', '');

            showModal('readerModal');

            

            document.getElementById('viewer').innerHTML = ""; // 清空舊內容

            ePubBook = ePub(fileUrl);

            rendition = ePubBook.renderTo("viewer", {

                width: "100%",

                height: "100%",

                spread: "none"

            });

            rendition.display();

        }


        function closeReader() {

            if(ePubBook) { ePubBook.destroy(); }

            closeModal('readerModal');

        }


        // --- ✏️ 分類修改邏輯 ---

        function openEditSeriesModal() {

            if(activeCategory === '全部書籍' || activeCategory === '未分類') {

                alert('「全部書籍」與「未分類」是系統預設，無法修改名稱。請點擊單一書籍右上角的齒輪來設定新分類。');

                return;

            }

            document.getElementById('oldSeriesName').value = activeCategory;

            document.getElementById('newSeriesName').value = activeCategory;

            showModal('editSeriesModal');

        }


        async function submitEditSeries() {

            const oldName = document.getElementById('oldSeriesName').value;

            const newName = document.getElementById('newSeriesName').value.trim();

            if(!newName || oldName === newName) return;


            const res = await fetch('/series/update', {

                method: 'POST',

                headers: {'Content-Type': 'application/json'},

                body: JSON.stringify({ old_name: oldName, new_name: newName })

            });

            if(res.ok) {

                closeModal('editSeriesModal');

                showToast("✅ 分類名稱修改成功");

                activeCategory = newName; // 更新目前所在分類

                loadBooks();

            }

        }


        // --- ⚙️ 單一書籍設定邏輯 ---

        function openBookSettings(event, bookId, currentCategory) {

            event.stopPropagation(); // 防止觸發打開閱讀器

            document.getElementById('settingBookId').value = bookId;

            document.getElementById('settingBookCategory').value = currentCategory === '未分類' ? '' : currentCategory;

            showModal('bookSettingsModal');

        }


        async function submitBookCategory() {

            const bookId = document.getElementById('settingBookId').value;

            let newCat = document.getElementById('settingBookCategory').value.trim();

            if(!newCat) newCat = '未分類';


            const res = await fetch('/books/update_category', {

                method: 'POST',

                headers: {'Content-Type': 'application/json'},

                body: JSON.stringify({ book_id: bookId, new_category: newCat })

            });

            if(res.ok) {

                closeModal('bookSettingsModal');

                showToast("✅ 書籍分類已更新");

                loadBooks();

            }

        }


        // --- 載入與渲染 ---

        async function loadBooks() {

            try {

                const res = await fetch('/books');

                if (res.ok) {

                    currentBooks = await res.json();

                    updateSeriesList();

                    renderBookGrid(activeCategory); // 保持在當前分類

                }

            } catch (e) { console.error(e); }

        }


        function updateSeriesList() {

            const list = document.getElementById('seriesList');

            const categories = ['全部書籍', ...new Set(currentBooks.map(b => b.series_name || '未分類'))];

            

            // 如果原本的分類被刪空了，退回全部書籍

            if(!categories.includes(activeCategory)) activeCategory = '全部書籍';


            list.innerHTML = categories.map(cat => `

                <li class="category-item ${cat === activeCategory ? 'active' : ''}" onclick="filterCategory(this, '${cat}')">${cat}</li>

            `).join('');

            document.getElementById('currentCategoryTitle').innerText = activeCategory;

        }


        function filterCategory(element, category) {

            activeCategory = category;

            document.querySelectorAll('.category-item').forEach(el => el.classList.remove('active'));

            element.classList.add('active');

            document.getElementById('currentCategoryTitle').innerText = category;

            renderBookGrid(category);

        }


        function renderBookGrid(category) {

            const grid = document.getElementById('bookGrid');

            let filtered = category === '全部書籍' ? currentBooks : currentBooks.filter(b => (b.series_name || '未分類') === category);

            

            document.getElementById('bookCount').innerText = `共 ${filtered.length} 本`;

            if (filtered.length === 0) {

                grid.innerHTML = `<div style="grid-column: 1/-1; text-align:center; color: #666; padding: 2rem;">目前沒有書籍資料</div>`;

                return;

            }

            

            // 點擊卡片閱讀，點擊齒輪設定分類

            grid.innerHTML = filtered.map(b => `

                <div class="book-card" onclick="openReader('${b.file_url}', '${b.title}')">

                    <i class="fa-solid fa-gear book-settings-btn" title="設定分類" onclick="openBookSettings(event, '${b.id}', '${b.series_name || '未分類'}')"></i>

                    <div class="book-title" title="${b.title}">${b.title.replace('.epub', '')}</div>

                    <div class="book-category">${b.series_name || '未分類'}</div>

                </div>

            `).join('');

        }


        // --- 系統共用邏輯 (上傳、登入等) ---

        async function handleFileSelect(files) {

            if (!userInfo.logged_in) { alert("請先登入"); return; }

            if (files.length === 0) return;

            document.getElementById('loading-overlay').style.display = 'flex';

            

            try {

                const formData = new FormData();

                for (let f of files) { if (f.name.endsWith('.epub')) formData.append('file', f); }

                const res = await fetch('/upload', { method: 'POST', body: formData });

                if(res.ok) { showToast("🎉 上傳成功！"); loadBooks(); }

            } finally {

                document.getElementById('loading-overlay').style.display = 'none';

                document.getElementById('fileInput').value = '';

            }

        }

        

        function toggleAuthMode() {

            isRegisterMode = !isRegisterMode;

            document.getElementById('modalTitle').innerText = isRegisterMode ? "註冊新帳號" : "會員登入";

            document.getElementById('toggleAuthText').innerText = isRegisterMode ? "已有帳號？前往登入" : "還沒有帳號？立即註冊";

        }


        async function handleAuthSubmit() {

            const u = document.getElementById('authUser').value, p = document.getElementById('authPass').value;

            if(!u || !p) return;

            const res = await fetch(isRegisterMode ? '/register' : '/login', {

                method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({username: u, password: p})

            });
